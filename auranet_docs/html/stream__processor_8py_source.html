43 if not ip_str:

48"""
49@brief Converts a Hubble event into AI features.
50
51Filters unsupported traffic and extracts the 13-dimensional
52network behavior feature vector.
53
54@param event Raw Hubble JSON event.
55@return NumPy feature array or None if ignored.
56"""
57 def process_event(self, event):
58 """
59 Filters and converts a single real-time packet into the 13-dim generic tensor.
60 Returns None if the packet should be ignored based on our training rules.
61 """
62 # SOFTWARE-LEVEL NODE FILTERING
63 event_node = event.get("node_name", "")
64 if event_node != config.NODE_NAME:
65 return None
66
67 flow = event.get("flow", {})
68 if not flow:
69 return None
70
71 verdict = flow.get("verdict", "")
72 source_labels = flow.get("source", {}).get("labels", [])
73 event_type = flow.get("event_type", {}).get("type", 0)
74 drop_reason = flow.get("drop_reason_desc", "")
75
76 if "reserved:host" in source_labels:
77 return None
78
79 # Prevent Double-Logging
80 if verdict == "DROPPED" and event_type != 1:
81 return None
82
83 # IGNORE SPIRE mTLS HANDSHAKE DROPS
84 if verdict == "DROPPED" and drop_reason == "AUTH_REQUIRED":
85 return None
86
87 l7 = flow.get("l7", {})
88
89 raw_port = flow.get("destination", {}).get("port")
90 try:
91 dst_port = int(raw_port) if raw_port else 0
92 except ValueError:
93 dst_port = 0
94
95 tcp_flags = flow.get("l4", {}).get("TCP", {}).get("flags", {})
96 is_syn = tcp_flags.get("SYN", False)
97 is_ack = tcp_flags.get("ACK", False)
98
99 # IGNORE L7 RESPONSES
100 if l7 and l7.get("type") == "RESPONSE":
101 return None
102
103 is_http_l7 = verdict == "FORWARDED" and l7 and "http" in l7
104 is_db_l4 = verdict == "FORWARDED" and dst_port == 5432 and is_syn and not is_ack
105
106 # Keep Drops, HTTP, OR Database TCP (L4)
107 if not (verdict == "DROPPED" or is_http_l7 or is_db_l4):
108 return None
109
110 traffic_direction = flow.get("traffic_direction", "UNKNOWN")
111 ip_source = flow.get("IP", {}).get("source", "")
112
113 http_data = l7.get("http", {}) if l7 else {}
114 method = http_data.get("method", "")
115 url = str(http_data.get("url", ""))
116 status_code = http_data.get("code", 0)
117
118 headers = http_data.get("headers", [])
119 header_size = sum(len(h.get("key", "")) + len(h.get("value", "")) for h in headers)
120
121 features = np.zeros(13, dtype=np.float32)
122
123 # 1. is_inbound
124 features[0] = 1.0 if traffic_direction == 'INGRESS' else 0.0
125
126 # 2. is_outbound
127 features[1] = 1.0 if traffic_direction == 'EGRESS' else 0.0
128
129 # 3. is_external_ip
130 features[2] = self.is_external_ip(ip_source)
131
132 # 4. is_dropped
133 features[3] = 1.0 if verdict == 'DROPPED' else 0.0
134
135 # 5. is_unknown_method
136 features[4] = 0.0 if method in self.valid_methods else 1.0
137
138 # 6. is_http
139 features[5] = 1.0 if is_http_l7 else 0.0
140
141 # 7. is_read_method
142 features[6] = 1.0 if method in self.read_methods else 0.0
143
144 # 8. is_write_method
145 features[7] = 1.0 if method in self.write_methods else 0.0
146
147 # 9. url_path_depth
148 path_depth = url.count('/')
149 features[8] = min(path_depth / self.MAX_PATH_DEPTH, 1.0)
150
151 # 10. query_param_count
152 param_count = url.count('&') + 1 if '?' in url else 0
153 features[9] = min(param_count / self.MAX_QUERY_PARAMS, 1.0)
154
155 # 11. req_header_size
156 features[10] = min(header_size / self.MAX_HEADER_SIZE, 1.0)
157
158 # 12. url_length
159 features[11] = min(len(url) / self.MAX_URL_LENGTH, 1.0)
160
161 # 13. is_error_status
162 features[12] = 1.0 if status_code >= 400 else 0.0
163
164 return features
