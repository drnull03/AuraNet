47 def load_and_filter(self):
48 print(f"🐝 Loading raw Hubble logs from {self.json_path}...")
49
50 filtered_events = []
51 with open(self.json_path, 'r') as f:
52 for line in f:
53 if not line.strip():
54 continue
55
56 try:
57 event = json.loads(line)
58 flow = event.get("flow", {})
59 except json.JSONDecodeError:
60 continue
61
62 verdict = flow.get("verdict", "")
63 source_labels = flow.get("source", {}).get("labels", [])
64 event_type = flow.get("event_type", {}).get("type", 0)
65 drop_reason = flow.get("drop_reason_desc", "")
66
67 if "reserved:host" in source_labels:
68 continue
69
70 if verdict == "DROPPED" and event_type != 1:
71 continue
72
73 if verdict == "DROPPED" and drop_reason == "AUTH_REQUIRED":
74 continue
75
76 l7 = flow.get("l7", {})
77 raw_port = flow.get("destination", {}).get("port")
78 try:
79 dst_port = int(raw_port) if raw_port else 0
80 except ValueError:
81 dst_port = 0
82
83 tcp_flags = flow.get("l4", {}).get("TCP", {}).get("flags", {})
84 is_syn = tcp_flags.get("SYN", False)
85 is_ack = tcp_flags.get("ACK", False)
86
87 if l7 and l7.get("type") == "RESPONSE":
88 continue
89
90 is_http_l7 = verdict == "FORWARDED" and l7 and "http" in l7
91 is_db_l4 = verdict == "FORWARDED" and dst_port == 5432 and is_syn and not is_ack
92
93 if verdict == "DROPPED" or is_http_l7 or is_db_l4:
94 self.raw_json_events.append(event)
95
96 src_app = self.extract_app_label(source_labels)
97 dst_app = self.extract_app_label(flow.get("destination", {}).get("labels", []))
98
99 http_data = l7.get("http", {}) if l7 else {}
100 method = http_data.get("method", "NONE")
101 url = http_data.get("url", "NONE")
102
103 filtered_events.append({
104 "verdict": verdict,
105 "src_app": src_app,
106 "dst_app": dst_app,
107 "method": method,
108 "url": url,
109 "dst_port": dst_port
110 })
111
112 print(f"✅ Filtered down to {len(filtered_events)} pure, relevant network events.")
113 self.raw_data = filtered_events

115 def engineer_features(self):
116 df = pd.DataFrame(self.raw_data)
117
118 # Target Label
119 df['is_dropped'] = (df['verdict'] == 'DROPPED').astype(float)
120
121 # Path Validations
122 acc_regex = re.compile(r"^.*/api/accounts\?id=[0-9]+$")
123 loan_regex = re.compile(r"^.*/api/loans/export\?id=L-[0-9]+$")
124
125 df['is_valid_acc_path'] = df['url'].apply(lambda x: 1.0 if acc_regex.match(str(x)) else 0.0)
126 df['is_valid_loan_path'] = df['url'].apply(lambda x: 1.0 if loan_regex.match(str(x)) else 0.0)
127
128 # Request Types
129 df['is_get'] = (df['method'] == 'GET').astype(float)
130 df['is_db_traffic'] = (df['dst_port'] == 5432).astype(float)
131
132 # Source Pod Matrix
133 df['src_is_frontend'] = (df['src_app'] == 'frontend-ui').astype(float)
134 df['src_is_gateway'] = (df['src_app'] == 'api-gateway').astype(float)
135 df['src_is_account'] = (df['src_app'] == 'account-service').astype(float)
136 df['src_is_loan'] = (df['src_app'] == 'loan-service').astype(float)
137
138 # Destination Pod Matrix
139 df['dst_is_gateway'] = (df['dst_app'] == 'api-gateway').astype(float)
140 df['dst_is_account'] = (df['dst_app'] == 'account-service').astype(float)
141 df['dst_is_loan'] = (df['dst_app'] == 'loan-service').astype(float)
142 df['dst_is_db'] = (df['dst_app'] == 'finance-db').astype(float)
143
144 self.dataframe = df.drop(columns=['verdict', 'src_app', 'dst_app', 'method', 'url', 'dst_port'])
145 print(f"🧠 Engineered Feature Matrix: {self.dataframe.shape[1]} dimensions.")
