1/**
2 * @file cleanup.js
3 * @brief AuraNet Bootstrap cleanup utility.
4 *
5 * Removes all network and runtime security policies previously
6 * deployed by the AuraNet Bootstrap microservice.
7 */
8const k8s = require('@kubernetes/client-node');
9
10const kc = new k8s.KubeConfig();
11kc.loadFromCluster();
12const customObjectsApi = kc.makeApiClient(k8s.CustomObjectsApi);
13/**
14 * Deletes all AuraNet-managed Cilium network policies and
15 * tracing policies from the Kubernetes cluster.
16 *
17 * Intended for development, testing, or complete environment
18 * teardown prior to redeployment.
19 *
20 * @async
21 * @returns {Promise<void>}
22 */
23async function wipePolicies() {
24 console.log("Starting AuraNet Teardown Sequence...");
25
26 // Wipe Cilium Network Policies
27 try {
28 const cnps = await customObjectsApi.listNamespacedCustomObject(
29 'cilium.io', 'v2', 'default', 'ciliumnetworkpolicies'
30 );
31 for (const item of cnps.body.items) {
32 await customObjectsApi.deleteNamespacedCustomObject(
33 'cilium.io', 'v2', 'default', 'ciliumnetworkpolicies', item.metadata.name
34 );
35 console.log(`[CLEANED] Deleted Network Policy: ${item.metadata.name}`);
36 }
37 } catch (e) {
38 const trueError = e.body ? JSON.stringify(e.body) : (e.cause ? e.cause.message : e.message);
39 console.error("[ERROR] wiping network policies:", trueError);
40 }
41
42 // Wipe Tracing PoliciesapplyRuntimePolicies
43 // because old naive policies were tracing policeis
44 try {
45 const tracingPolicies = await customObjectsApi.listNamespacedCustomObject(
46 'cilium.io', 'v1alpha1', 'default', 'tracingpoliciesnamespaced'
47 );
48 for (const item of tracingPolicies.body.items) {
49 await customObjectsApi.deleteNamespacedCustomObject(
50 'cilium.io', 'v1alpha1', 'default', 'tracingpoliciesnamespaced', item.metadata.name
51 );
52 console.log(`[CLEANED] Deleted Tracing Policy: ${item.metadata.name}`);
53 }
54 } catch (e) {
55 const trueError = e.body ? JSON.stringify(e.body) : (e.cause ? e.cause.message : e.message);
56 console.error("[ERROR] wiping tracing policies:", trueError);
57 }
58}
