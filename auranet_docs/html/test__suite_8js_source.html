13function parseTetragonEvent(eventJson, mockPublishFn) {
14 try {
15 const event = JSON.parse(eventJson);
16 if (!event.process_kprobe) return;
17
18 const processData = event.process_kprobe.process;
19 const functionName = event.process_kprobe.function_name;
20
21 if (!processData || !processData.pod) return;
22
23 const namespace = processData.pod.namespace;
24 const labels = processData.pod.labels || {};
25 const workload = labels['app'] || labels['k8s-app'] || 'unknown-workload';
26
27 // Ignore system namespaces
28 if (namespace === 'kube-system' || namespace === 'auranet-namespace') return;
29
30 let threatSignature = null;
31 let actionContext = '';
32
33 if (functionName === 'sys_execve') {
34 const binary = processData.binary.split('/').pop();
35 threatSignature = THREAT_MAP[binary];
36 actionContext = `Executed binary: ${binary}`;
37 } else if (functionName === 'security_file_open') {
38 const args = event.process_kprobe.args || [];
39 if (args.length > 0 && args[0].file_arg) {
40 const filePath = args[0].file_arg.path;
41 const fileName = filePath.split('/').pop();
42 threatSignature = THREAT_MAP[filePath] || THREAT_MAP[fileName] ||
43 (filePath.includes('token') ? 'unauthorized_file_read' : null);
44 actionContext = `Accessed file: ${filePath}`;
45 }
46 }
47
48 if (threatSignature) {
49 const subject = `auranet.events.runtime.${workload}`;
50 const payload = { source: "runtime_ebpf", threat: threatSignature, context: actionContext };
51
52 // Trigger the mock instead of the real network!
53 mockPublishFn(subject, payload);
54 }
55 } catch (err) {
56 // Silently ignore malformed JSON just like the real forwarder
57 }
58}
