18"""
19Injects a trusted eBPF identity into the AuraNet trust matrix.
20
21Updates the AuraNet ConfigMap and triggers propagation of the new
22trusted identity configuration to edge agents.
23
24@param label Kubernetes/eBPF identity label to trust.
25@param namespace Kubernetes namespace containing AuraNet resources.
26@returns None
27"""
28def inject_trusted_label(label: str, namespace: str = "auranet-namespace"):
29 """
30 Connects to the K8s API, modifies the AuraNet ConfigMap in memory,
31 and patches the live cluster to trigger a hot-reload on all edge nodes.
32 """
33 try:
34 config.load_kube_config()
35 v1 = client.CoreV1Api()
36 config_map_name = "auranet-engine-config"
37
38 print(f"Connecting to Kubernetes API in namespace '{namespace}'...")
39
40 cm = v1.read_namespaced_config_map(name=config_map_name, namespace=namespace)
41 ai_config_str = cm.data.get('ai-config.json', '{}')
42 ai_config = json.loads(ai_config_str)
43
44 if "trustedIdentities" not in ai_config:
45 ai_config["trustedIdentities"] = []
46
47 if label in ai_config["trustedIdentities"]:
48 print(f"⚠️ The eBPF label '{label}' is already in the trust matrix. Exiting.")
49 sys.exit(0)
50
51 ai_config["trustedIdentities"].append(label)
52
53 cm.data['ai-config.json'] = json.dumps(ai_config, indent=2)
54 v1.patch_namespaced_config_map(name=config_map_name, namespace=namespace, body=cm)
55
56 print(f"✅ Successfully injected trusted identity: {label}")
57 print(f"Kubernetes is propagating the update to all AuraNet edge agents. Hot-reload imminent.")
58
59 except Exception as e:
60 print(f"❌ Failed to update the cluster: {e}")
61 sys.exit(1)

64"""

67Revokes previously granted trust permissions by updating the

74def remove_trusted_label(label: str, namespace: str = "auranet-namespace"):
75 """
76 Removes an existing eBPF label from the ConfigMap trust matrix to revoke immunity.
77 """
78 try:
79 config.load_kube_config()
80 v1 = client.CoreV1Api()

83 print(f"Connecting to Kubernetes API in namespace '{namespace}'...")
84

86 ai_config_str = cm.data.get('ai-config.json', '{}')
87 ai_config = json.loads(ai_config_str)
88
89 if "trustedIdentities" not in ai_config or label not in ai_config["trustedIdentities"]:
90 print(f"The eBPF label '{label}' is not currently in the trust matrix. Exiting.")
91 sys.exit(0)
92
93 # Remove the immunity label
94 ai_config["trustedIdentities"].remove(label)
95

97 v1.patch_namespaced_config_map(name=config_map_name, namespace=namespace, body=cm)
98
99 print(f" Successfully revoked trusted identity: {label}")
100 print(f"Kubernetes is propagating the update. The AI will now evaluate this workload.")
101
102 except Exception as e:
103 print(f"❌ Failed to update the cluster: {e}")

117def _run(cmd: list[str]):
118 """Run a subprocess command, streaming output, and exit on failure."""
