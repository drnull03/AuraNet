44# @brief Loads Hubble logs and filters relevant security events.
45#
46# @details
47# Removes irrelevant traffic and keeps events related to forwarded,
48# dropped, HTTP, and database connection behaviour.
49#
50 def load_and_filter(self):
51 print(f"🐝 Loading raw Hubble logs from {self.json_path}...")
52
53 filtered_events = []
54 with open(self.json_path, 'r') as f:
55 for line in f:
56 if not line.strip():
57 continue
58
59 try:
60 event = json.loads(line)
61 flow = event.get("flow", {})
62 except json.JSONDecodeError:
63 continue
64
65 verdict = flow.get("verdict", "")
66 source_labels = flow.get("source", {}).get("labels", [])
67 event_type = flow.get("event_type", {}).get("type", 0)
68 drop_reason = flow.get("drop_reason_desc", "")
69
70 if "reserved:host" in source_labels:
71 continue
72
73 if verdict == "DROPPED" and event_type != 1:
74 continue
75
76 if verdict == "DROPPED" and drop_reason == "AUTH_REQUIRED":
77 continue
78
79 l7 = flow.get("l7", {})
80
81 raw_port = flow.get("destination", {}).get("port")
82 try:
83 dst_port = int(raw_port) if raw_port else 0
84 except ValueError:
85 dst_port = 0
86
87 tcp_flags = flow.get("l4", {}).get("TCP", {}).get("flags", {})
88 is_syn = tcp_flags.get("SYN", False)
89 is_ack = tcp_flags.get("ACK", False)
90
91 if l7 and l7.get("type") == "RESPONSE":
92 continue
93
94 is_http_l7 = verdict == "FORWARDED" and l7 and "http" in l7
95 is_db_l4 = verdict == "FORWARDED" and dst_port == 5432 and is_syn and not is_ack
96
97 if not (verdict == "DROPPED" or is_http_l7 or is_db_l4):
98 continue
99
100 traffic_direction = flow.get("traffic_direction", "UNKNOWN")
101 ip_source = flow.get("IP", {}).get("source", "")
102
103 http_data = l7.get("http", {}) if l7 else {}
104 method = http_data.get("method", "")
105 url = http_data.get("url", "")
106 status_code = http_data.get("code", 0)
107
108 headers = http_data.get("headers", [])
109 header_size = sum(len(h.get("key", "")) + len(h.get("value", "")) for h in headers)
110
111 self.raw_json_events.append(event)
112
113 filtered_events.append({
114 "traffic_direction": traffic_direction,
115 "ip_source": ip_source,
116 "verdict": verdict,
117 "is_http": 1 if is_http_l7 else 0,
118 "method": method,
119 "url": url,
120 "status_code": status_code,
121 "header_size": header_size
122 })
123
124 print(f"✅ Filtered down to {len(filtered_events)} pure, relevant network events.")
125 self.raw_data = filtered_events
126##
127# @brief Determines whether an IP address belongs to external traffic.
128#
129# @param ip_str Source IP address.
130#
131# @return 1.0 if external, otherwise 0.0.
132#
133 def is_external_ip(self, ip_str):
134 if not ip_str: return 0.0
135 if ip_str.startswith("10.") or ip_str.startswith("192.168.") or ip_str.startswith("172."):
136 return 0.0
137 return 1.0

140# @brief Converts filtered telemetry into numerical ML features.
141#
142# @details
143# Performs feature extraction, normalization, and encoding of
144# network behaviour into a universal feature matrix.
145#
146 def engineer_features(self):
147 df = pd.DataFrame(self.raw_data)
148
149 # --- L3 Network Baseline ---
150 df['is_inbound'] = (df['traffic_direction'] == 'INGRESS').astype(float)
151 df['is_outbound'] = (df['traffic_direction'] == 'EGRESS').astype(float)
152 df['is_external_ip'] = df['ip_source'].apply(self.is_external_ip)
153
154 # --- Zero Trust Policy ---
155 df['is_dropped'] = (df['verdict'] == 'DROPPED').astype(float)
156
157 valid_methods = ['GET', 'POST', 'PUT', 'DELETE', 'PATCH', 'HEAD', 'OPTIONS', '']
158 df['is_unknown_method'] = (~df['method'].isin(valid_methods)).astype(float)
159
160 # --- L7 HTTP Structural Metadata ---
161 df['is_http'] = df['is_http'].astype(float)
162
163 df['is_read_method'] = df['method'].isin(['GET', 'OPTIONS', 'HEAD']).astype(float)
164 df['is_write_method'] = df['method'].isin(['POST', 'PUT', 'DELETE', 'PATCH']).astype(float)
165
166 df['url_path_depth'] = df['url'].apply(lambda x: str(x).count('/'))
167 df['query_param_count'] = df['url'].apply(lambda x: str(x).count('&') + 1 if '?' in str(x) else 0)
168
169 df['url_path_depth'] = np.clip(df['url_path_depth'] / self.MAX_PATH_DEPTH, 0.0, 1.0)
170 df['query_param_count'] = np.clip(df['query_param_count'] / self.MAX_QUERY_PARAMS, 0.0, 1.0)
171 df['req_header_size'] = np.clip(df['header_size'] / self.MAX_HEADER_SIZE, 0.0, 1.0)
172
173 # REPLACED req_body_size WITH url_length
174 df['url_length'] = df['url'].apply(lambda x: len(str(x)))
175 df['url_length'] = np.clip(df['url_length'] / self.MAX_URL_LENGTH, 0.0, 1.0)
176
177 df['is_error_status'] = (df['status_code'] >= 400).astype(float)
178
179 cols_to_drop = ['traffic_direction', 'ip_source', 'verdict', 'method', 'url', 'status_code', 'header_size']
180 self.dataframe = df.drop(columns=cols_to_drop)
181
182 print(f"🧠 Engineered Universal Feature Matrix: {self.dataframe.shape[1]} dimensions.")
