19/**
20 * Wraps Node Fetch with a retry mechanism to handle Kubernetes pod cycling
21 * and temporary port-forward tunnel drops.

23async function fetchWithRetry(url, retries = 5, delayMs = 2000) {
24 for (let i = 0; i < retries; i++) {
25 try {
26 const res = await fetch(url);
27 return res; // Success!
28 } catch (err) {
29 if (i === retries - 1) throw err; // Out of retries
30 console.log(`[Network] Tunnel syncing or pod booting. Retrying in ${delayMs/1000}s...`);
31 await new Promise(resolve => setTimeout(resolve, delayMs));
32 }
33 }

36/**
37 * Robust asynchronous observer that captures the next message on a wildcard subject
38 * matching a specific internal criteria function.

40function listenForEvent(subjectWildcard, criteriaFn, timeoutMs = 8000) {
41 return new Promise((resolve, reject) => {
42 const sub = nc.subscribe(subjectWildcard);
43
44 const timeout = setTimeout(() => {
45 sub.unsubscribe();
46 reject(new Error(`⏳ Timeout: Did not match expected event stream on ${subjectWildcard}`));
47 }, timeoutMs);
48
49 (async () => {
50 for await (const msg of sub) {
51 try {
52 const payload = JSON.parse(sc.decode(msg.data));
53 if (criteriaFn(payload, msg.subject)) {
54 clearTimeout(timeout);
55 sub.unsubscribe();
56 resolve(payload);
57 break;
58 }
59 } catch (e) {
60 // Ignore malformed JSON or parsing errors from other test noise
61 }
62 }
63 })();
64 });
