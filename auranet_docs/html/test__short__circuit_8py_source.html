38class DummyBrainA(nn.Module):
39 def __init__(self, force_anomaly=False):
40 super().__init__()
41 self.force_anomaly = force_anomaly
42 def forward(self, x):
43 # Adding 10.0 forces a massive MSE Loss. Returning x forces 0.0 MSE Loss.
44 return x + 10.0 if self.force_anomaly else x
45
46# Dummy Brain B & C (NLP Transformers/LSTMs)
47class DummyBrainNLP(nn.Module):
48 def __init__(self, force_anomaly=False):
49 super().__init__()
50 self.force_anomaly = force_anomaly
51 def forward(self, x):
52 batch, seq_len = x.shape
53 logits = torch.zeros(batch, seq_len, 128)
54 if self.force_anomaly:
55 # Force high CrossEntropy loss by predicting completely wrong chars
56 logits[:, :, 0] = 10.0
57 else:
58 # Force near-zero CE loss by correctly predicting the input chars
59 for b in range(batch):
60 for s in range(seq_len):
61 logits[b, s, x[b, s]] = 10.0
62 return logits
